import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (i == j || i + j == n + 1)   // X shape condition
                    System.out.print("* ");
                else
                    System.out.print("  ");
            }
            System.out.println();
        }
    }
}

import java.util.*;//DIAMOND
public class Main
{
	public static void main(String[] args) {
	    Scanner sc = new Scanner(System.in);
	    int n = sc.nextInt();
	    for(int i = 1;i<n;i++){
	        for(int k = i;k<n;k++){
	            System.out.print(" ");
	        }
	            for(int j =1;j<=2*i-1;j++){
	               System.out.print("*");
	            }
	            System.out.println();
	             
	        }
	        for(int i = n-1;i>=1;i--){
	        for(int k = i;k<n;k++){
	            System.out.print(" ");
	        }
	            for(int j =1;j<=2*i-1;j++){
	               System.out.print("*");
	            }
	            System.out.println();
	             
	        }
	}
}
